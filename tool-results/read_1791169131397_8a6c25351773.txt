   180→function LoginCard({ onLoginSuccess }: { onLoginSuccess: () => void }) {
   181→  const [password, setPassword] = React.useState("");
   182→  const [loading, setLoading] = React.useState(false);
   183→  const [error, setError] = React.useState<string | null>(null);
   184→
   185→  const submit = async (e: React.FormEvent) => {
   186→    e.preventDefault();
   187→    setLoading(true);
   188→    setError(null);
   189→    try {
   190→      const res = await fetch("/api/admin/login", {
   191→        method: "POST",
   192→        headers: { "Content-Type": "application/json" },
   193→        body: JSON.stringify({ password }),
   194→      });
   195→      const data = await res.json().catch(() => ({}));
   196→      if (!res.ok) {
   197→        setError(data?.error || "ورود ناموفق بود");
   198→        toast.error(data?.error || "ورود ناموفق بود");
   199→        return;
   200→      }
   201→      toast.success("خوش آمدید");
   202→      onLoginSuccess();
   203→    } catch {
   204→      setError("ارتباط با سرور ناموفق بود");
   205→      toast.error("ارتباط با سرور ناموفق بود");
   206→    } finally {
   207→      setLoading(false);
   208→    }
   209→  };
   210→
   211→  return (
   212→    <motion.div
   213→      initial={{ opacity: 0, y: 8 }}
   214→      animate={{ opacity: 1, y: 0 }}
   215→      transition={{ duration: 0.2 }}
   216→      className="flex justify-center py-10"
   217→    >
   218→      <Card className="w-full max-w-md shadow-md">
   219→        <CardHeader>
   220→          <div className="mb-2 flex h-12 w-12 items-center justify-center rounded-xl bg-primary text-primary-foreground">
   221→            <ShieldCheck className="h-6 w-6" />

   327→function UploadCard({
   328→  groups,
   329→  onUploaded,
   330→}: {
   331→  groups: AdminGroup[];
   332→  onUploaded: () => void;
   333→}) {
   334→  const [file, setFile] = React.useState<File | null>(null);
   335→  const [displayName, setDisplayName] = React.useState("");
   336→  const [slug, setSlug] = React.useState("");
   337→  const [description, setDescription] = React.useState("");
   338→  const [groupId, setGroupId] = React.useState<string>("none");
   339→  const [uploading, setUploading] = React.useState(false);
   340→
   341→  const reset = () => {
   342→    setFile(null);
   343→    setDisplayName("");
   344→    setSlug("");
   345→    setDescription("");
   346→    setGroupId("none");
   347→  };
   348→
   349→  const onFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   350→    const f = e.target.files?.[0] ?? null;
   351→    setFile(f);
   352→    if (f && !displayName) {
   353→      // pre-fill displayName from filename (strip extension)
   354→      setDisplayName(f.name.replace(/\.mp4$/i, ""));
   355→    }
   356→  };
   357→
   358→  const submit = async (e: React.FormEvent) => {
   359→    e.preventDefault();
   360→    if (!file) {
   361→      toast.error("فایل ویدئو را انتخاب کنید");
   362→      return;
   363→    }
   364→    if (!displayName.trim()) {
   365→      toast.error("نام نمایشی را وارد کنید");
   366→      return;
   367→    }
   368→    if (!file.name.toLowerCase().endsWith(".mp4")) {
   369→      toast.error("فقط فایل‌های mp4 پذیرفته می‌شوند");
   370→      return;
   371→    }
   372→    setUploading(true);
   373→    try {
   374→      const form = new FormData();
   375→      form.append("file", file);
   376→      form.append("displayName", displayName.trim());
   377→      if (slug.trim()) form.append("slug", slug.trim());
   378→      if (description.trim()) form.append("description", description.trim());
   379→      if (groupId && groupId !== "none")
   380→        form.append("groupId", groupId);
   381→      const res = await fetch("/api/admin/videos", {
   382→        method: "POST",
   383→        body: form,
   384→      });
   385→      const data = await res.json().catch(() => ({}));
   386→      if (!res.ok) {
   387→        toast.error(data?.error || "آپلود ناموفق بود");
   388→        return;
   389→      }
   390→      toast.success("ویدئو با موفقیت آپلود شد");
   391→      reset();
   392→      onUploaded();
   393→    } catch {
   394→      toast.error("ارتباط با سرور ناموفق بود");
   395→    } finally {
   396→      setUploading(false);
   397→    }
   398→  };
   399→
   400→  return (
   401→    <Card>
   402→      <CardHeader>
   403→        <CardTitle className="flex items-center gap-2 text-base">
   404→          <Upload className="h-5 w-5" />
   405→          آپلود ویدئوی جدید
   406→        </CardTitle>
   407→        <CardDescription>
   408→          فایل را انتخاب کنید و اطلاعات را تکمیل کنید.
   409→        </CardDescription>
   410→      </CardHeader>
   411→      <CardContent>
   412→        <form onSubmit={submit} className="grid gap-4 sm:grid-cols-2">
   413→          <div className="space-y-2 sm:col-span-2">
   414→            <Label htmlFor="file">فایل ویدئو (mp4)</Label>
   415→            <Input
   416→              id="file"
   417→              type="file"
   418→              accept="video/mp4"
   419→              onChange={onFileChange}
   420→              disabled={uploading}
   421→            />
   422→            {file ? (
   423→              <p className="text-xs text-muted-foreground">
   424→                فایل: <span dir="ltr">{file.name}</span> —{" "}
   425→                {formatBytes(file.size)}
   426→              </p>
   427→            ) : null}
   428→          </div>
   429→          <div className="space-y-2">
   430→            <Label htmlFor="displayName">نام نمایشی</Label>
   431→            <Input
   432→              id="displayName"
   433→              value={displayName}
   434→              onChange={(e) => setDisplayName(e.target.value)}
   435→              placeholder="مثلاً: درس اول"
   436→              disabled={uploading}
   437→            />
   438→          </div>
   439→          <div className="space-y-2">
   440→            <Label htmlFor="slug">اسلاگ (اختیاری)</Label>
   441→            <Input
   442→              id="slug"
   443→              value={slug}
   444→              onChange={(e) => setSlug(e.target.value)}
   445→              placeholder="خودکار از نام"
   446→              disabled={uploading}
   447→              dir="ltr"
   448→            />
   449→          </div>
   450→          <div className="space-y-2">
   451→            <Label htmlFor="group">گروه</Label>
   452→            <Select value={groupId} onValueChange={setGroupId} disabled={uploading}>
   453→              <SelectTrigger id="group" className="w-full">
   454→                <SelectValue placeholder="بدون گروه" />
   455→              </SelectTrigger>
   456→              <SelectContent>
   457→                <SelectItem value="none">بدون گروه</SelectItem>
   458→                {groups.map((g) => (
   459→                  <SelectItem key={g.id} value={g.id}>
   460→                    {g.name}
   461→                  </SelectItem>
   462→                ))}
   463→              </SelectContent>
   464→            </Select>
   465→          </div>
   466→          <div className="space-y-2 sm:col-span-2">
   467→            <Label htmlFor="description">توضیحات (اختیاری)</Label>
   468→            <Textarea
   469→              id="description"
   470→              value={description}
   471→              onChange={(e) => setDescription(e.target.value)}
   472→              placeholder="توضیح کوتاه درباره ویدئو..."
   473→              disabled={uploading}
   474→              className="min-h-20"
   475→            />
   476→          </div>
   477→          <div className="sm:col-span-2">
   478→            <Button type="submit" disabled={uploading} className="gap-1.5">
   479→              {uploading ? (
   480→                <>
   481→                  <Loader2 className="h-4 w-4 animate-spin" />
   482→                  در حال آپلود...
   483→                </>
   484→              ) : (
   485→                <>
   486→                  <Upload className="h-4 w-4" />
   487→                  آپلود
   488→                </>
   489→              )}
   490→            </Button>
   491→          </div>
   492→        </form>
   493→      </CardContent>
   494→    </Card>
   495→  );
   496→}

   565→        }}
   566→      />
   567→    </div>
   568→  );

   571→function VideoRow({
   572→  v,
   573→  groups,
   574→  onEdit,
   575→  onDelete,
   576→}: {
   577→  v: AdminVideo;
   578→  groups: AdminGroup[];
   579→  onEdit: () => void;
   580→  onDelete: () => void;
   581→}) {
   582→  const [copied, setCopied] = React.useState(false);
   583→  const origin =
   584→    typeof window !== "undefined" ? window.location.origin : "";
   585→  const link = `${origin}/?v=${encodeURIComponent(v.slug)}`;
   586→
   587→  const copyLink = async () => {
   588→    try {
   589→      await navigator.clipboard.writeText(link);
   590→      setCopied(true);
   591→      toast("لینک کپی شد");
   592→      setTimeout(() => setCopied(false), 1500);
   593→    } catch {
   594→      toast.error("کپی ناموفق بود");
   595→    }
   596→  };
   597→
   598→  return (
   599→    <Card>
   600→      <CardContent className="p-4">
   601→        <div className="flex flex-col gap-3 sm:flex-row sm:items-start">
   602→          <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-muted text-muted-foreground">
   603→            <Film className="h-5 w-5" />
   604→          </div>
   605→          <div className="min-w-0 flex-1 space-y-1.5">
   606→            <div className="flex flex-wrap items-center gap-2">
   607→              <h4 className="text-sm font-semibold leading-tight">
   608→                {v.displayName}
   609→              </h4>
   610→              <Badge variant="secondary" className="text-[10px]" dir="ltr">
   611→                {v.slug}
